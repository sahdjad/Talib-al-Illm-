\subsection*{Fragen zur Einführung und zur Exegese}

\begin{enumerate}[leftmargin=1.6em]
\item Wer ist der Verfasser des Buches „Die wichtigen Lektionen“?
  \begin{itemize}
    \item ʿAbdul ʿAzīz Ibn Bāz, möge Allah Sich seiner erbarmen
    \item Moḥāmmad Ibn ʿUṯaymīn, möge Allah Sich seiner erbarmen
    \item Hayṯam Sarḥān, möge Allah ihn beschützen
  \end{itemize}
\item Warum lernen wir die wichtigen Lektionen?
  \begin{itemize}
    \item weil sie wichtig sind.
    \item weil die Gelehrten empfahlen, sich damit zu beschäftigen.
    \item weil sie jene wichtigen Angelegenheiten umfassen, die der Muslim braucht.
    \item wegen all der vorigen Aspekte.
  \end{itemize}
\item Dieses Buch umfasst:
  \begin{itemize}
    \item die Beziehung des Muslims zum Qur’ān und zum monotheistischen Glauben.
    \item das Gebet und die rituelle Gebetswaschung (wuḍū’).
    \item die Sitten und Moralen.
    \item die Warnung vor den Sünden.
    \item die Bestattung des Verstorbenen.
    \item all die vorigen Aspekte.
  \end{itemize}
\item In seiner Beschäftigung mit dem Qur’ān beginnt der Muslim sowohl beim Erlernen der
  Rezitation, bei derer Korrektur, beim Auswendiglernen und beim Erlernen der Exegese mit:
  \begin{itemize}
    \item Sūrat al-ʿAlaq (96: Das Anhängsel)
    \item Sūrat al-Fātiḥa (1: Die Eröffnung)
    \item Sūrat al-Iḫlāṣ (112: Die Aufrichtigkeit)
  \end{itemize}
\item Gemäß ihrem Verhalten gegenüber dem Qur’ān lassen sich die Menschen in zwei Extremen
  und einer Mitte einteilen. (richtig -- falsch)
\item Welches Buch der Exegese soll ein Studierender zuerst lesen?
  \begin{itemize}
    \item Ibn Kaṯīr
    \item Ibn as-Saʿdiyy
    \item al-Qurṭubī
  \end{itemize}
\item Ein Studierender soll mit den kurzen Erläuterungen anfangen, dann zu den längeren
  übergehen. (richtig -- falsch)
\item Der Studierende soll in seiner Beschäftigung mit den Exegese-Büchern mit jener Sūra
  anfangen, die ihn zur Fortsetzung der Lektüre motiviert und die man oft zu vernehmen bekommt,
  wie Sūrat al-Qaṣaṣ (28), Maryam (19), al-Kahf (18). (richtig -- falsch)
\item Der Studierende könnte sich die Exegese-Bücher in Form von Hörbüchern anhören, wenn
  ihm das Lesen schwerfällt. (richtig -- falsch)
\item Der Prophet \ar{ﷺ} warnte davor, dass man den Qur’ān rezitiert, ohne sich über seine
  Bedeutung Gedanken zu machen. (richtig -- falsch)
\end{enumerate}

\subsection*{Fragen zur Exegese von Sūrat al-Fātiḥa (1: Die Eröffnung)}

\begin{enumerate}[leftmargin=1.6em]
\setcounter{enumi}{10}
\item Eine Sūra erhielt diese Bezeichnung, weil sie (so komplett ist, als ob sie) mit einem Zaun
  (arab. \emph{sūr}) versehen wäre, d.\,h. sie darf weder etwas einbüßen, noch darf etwas
  hinzugefügt werden. (richtig -- falsch)
\item Die erste Sūra erhielt die Bezeichnung „al-Fātiḥa“ (Die Eröffnung), weil:\\
  \hrulefill\\
  \hrulefill
\item Sūrat al-Fātiḥa wird auch genannt:
  \begin{itemize}
    \item Mutter des Qur’āns (\emph{umm al-Qur’ān})
    \item die sieben Lobverse (\emph{as-sabʿ al-maṯānī})
    \item die Heilung (\emph{ar-ruqyah})
    \item das Gebet (\emph{aṣ-ṣalāh})
    \item all die vorigen Bezeichnungen
  \end{itemize}
\item Es ist eine Pflicht, vor der Rezitation des Qur’āns Allah um Zuflucht vor dem Teufel zu
  bitten, obwohl die Rezitation als eine Anbetung Gottes und nicht als Sünde gilt. Weswegen?\\
  \hrulefill\\
  \hrulefill
\item Was bedeutet „ich bitte um Zuflucht“ (\emph{aʿūḏu})?\\
  \hrulefill
\item Der Teufel wird als der Gesteinigte (\emph{ar-rağīm}) bezeichnet, \ldots
  \begin{itemize}
    \item weil er aus der Barmherzigkeit (Gottes) ausgeschlossen wurde.
    \item weil er mit Meteoriten gesteinigt wird.
    \item weil er die Menschen mit Leidenschaften und Zweifeln besteinigt.
    \item aus allen vorigen Gründen.
  \end{itemize}
\item Die Präpositionalkonstruktion in der Formel „bi-’smi-’llāh“ (im Namen Allahs) ist eine
  Modalergänzung, die mit einem ausgelassenen, nachgestellten Prädikat zusammenhängt.
  (richtig -- falsch)
\item Allah:
  \begin{itemize}
    \item Er ist der aus Liebe und Ehrfurcht angebetete Gott.
    \item Niemandem wurde dieser Name gegeben außer Allah.
    \item Das Bezugswort für alle (99 schönen) Namen Gottes.
    \item Man sagt, dass dieser der großartigste aller Namen Gottes ist.
    \item Der bestimmte Artikel „Al“ wird im Appellativ nicht ausgelassen.
    \item Dafür gelten all die vorigen Aspekte.
  \end{itemize}
\item Was ist der Unterschied zwischen den beiden schönen Namen „ar-Raḥmān“ (dem Erbarmer)
  und „ar-Raḥīm“ (dem Barmherzigen)?\\
  \hrulefill\\
  \hrulefill
\item Die Herrschaft Allahs, erhaben sei Er, über Seine Geschöpfe ist in zwei Arten einzuteilen,
  welche?
  \begin{itemize}
    \item eine allgemeine und eine spezifische.
    \item eine relative und eine absolute.
  \end{itemize}
\item Die meisten Bittgebete der Propheten, Allahs Frieden auf ihnen, werden eingeleitet mit dem
  Wort:
  \begin{itemize}
    \item „Allāhumm“ (O Allah)
    \item „ar-Rabb“ (Der Gott)
  \end{itemize}
\item Der Tag des Gerichts ist:
  \begin{itemize}
    \item Der Tag der Auferstehung.
    \item Der Tag, an dem die Menschen gemäß ihren Taten zur Rechenschaft gezogen werden.
    \item Alle vorigen Aspekte.
  \end{itemize}
\item „Leite uns den geraden Weg“ (1:6) ist das allumfassendste Bittgebet und das dem Menschen
  am nützlichsten. (richtig -- falsch)
\item Das Wort „ad-dīn“ (das Gericht, die Gerechtigkeit) bezeichnet:
  \begin{itemize}
    \item die Vergeltung.
    \item die Taten.
    \item sowohl die Vergeltung als auch die Taten.
  \end{itemize}
\item Dass das Patiens dem Agens vorangestellt wird, \ldots
  \begin{itemize}
    \item dient der Einschränkung.
    \item hat keine Funktion.
  \end{itemize}
\item Dass die Anbetung dem Ruf um Hilfe vorgezogen wird, gilt als Voranstellung:
  \begin{itemize}
    \item des Generellen vor dem Spezifischen.
    \item der Verpflichtung gegenüber Allah, erhaben sei Er, im Vergleich zum Recht Seiner
      Diener.
    \item beide Aspekte.
  \end{itemize}
\item Warum steht das Agens in dem Vers „Dir allein dienen wir“ (1:5) in der Mehrzahl?\\
  \hrulefill\\
  \hrulefill
\item Die Anbetung:
  \begin{itemize}
    \item ist eine umfassende Bezeichnung für alles, was Allah liebt und was Ihm an
      offenkundigen und verborgenen Worten und Taten wohlgefällt.
    \item ist die Unterwürfigkeit Allah gegenüber, indem man aus Liebe und Ehrfurcht Seine
      Gebote vollbringt und Seine Verbote vermeidet.
    \item das bezeichnet sowohl das eine als auch das andere.
  \end{itemize}
\item Die gemeinte Rechtleitung in dem Ausdruck „Leite uns“ (1:6) bezeichnet:
  \begin{itemize}
    \item Zeige uns (den richtigen Weg) und führe uns dorthin.
    \item Lass den Erfolg uns zuteilwerden.
    \item Beides.
  \end{itemize}
\item Diejenigen, auf die sich der Ausdruck „den Weg derjenigen, denen Du Gunst erwiesen hast“
  (1:7) bezieht, sind:
  \begin{itemize}
    \item Alle Gläubigen in dieser Gemeinschaft.
    \item Alle Propheten, aufrichtigen Gefährten, Märtyrer und Rechtschaffenen, denen Allah
      Seine Gunst erwiesen hat.
  \end{itemize}
\item Der Vers „Dir allein dienen wir“ (1:5) umfasst sowohl das islamische Recht als auch das
  Schicksal. (richtig -- falsch)
\item Welche Sūra impliziert das, was keine andere Sūra im Qur’ān enthält?
  \begin{itemize}
    \item Sūrat al-Fātiḥa (1)
    \item Āyat al-Kursiyy (2:255)
    \item Sūrat al-Iḫlāṣ (112)
  \end{itemize}
\item Der Ausdruck „Leite uns den [\ldots{}] Weg“ (1:6) impliziert:
  \begin{itemize}
    \item Die Bestätigung, dass der Prophet Moḥāmmad Allahs Gesandter ist.
    \item Eine Antwort auf alle Ketzer und Irreführer.
    \item Beide Aspekte.
  \end{itemize}
\item Der Ausdruck „dem Herrscher am Tag des Gerichts“ (1:4) bedeutet, \ldots
  \begin{itemize}
    \item dass die Vergeltung gerecht sein wird.
    \item dass der Mensch ein wahrer Urheber seiner Taten ist.
    \item beide vorigen Aspekte.
    \item dass die Vergeltung nur auf Taten basiert.
  \end{itemize}
\end{enumerate}

\subsection*{Fragen zur Exegese von „Āyat al-Kursiyy“ (2:255, dem Thronvers)}

\begin{enumerate}[leftmargin=1.6em,resume]
\item Dieser Vers wurde als der „Thronvers“ bezeichnet, weil darin der Thron (Gottes) erwähnt
  wird. (richtig -- falsch)
\item Was ist der großartigste Vers im Qur’ān?
  \begin{itemize}
    \item Der Schuldvers (2:282).
    \item Der Zehnrechte-Vers (4:36).
    \item Der Thronvers (2:255).
  \end{itemize}
\item Der Qur’ān ist semantisch gesehen meisterhaft. (richtig -- falsch)
\item Wie viele der schönen Namen Allahs enthält der Thronvers?
  \begin{itemize}
    \item Fünf. \item Sechs. \item Sieben.
  \end{itemize}
\item Allahs Name al-Ḥayy (Der ewig Lebende) ist ein Zeichen der Vollkommenheit; diese bezieht
  sich auf:
  \begin{itemize}
    \item Sein Selbst. \item Seine Herrschaft.
  \end{itemize}
\item Allahs Name al-Qayyūm (Der Beständige) ist ein Zeichen der Vollkommenheit; diese bezieht
  sich auf:
  \begin{itemize}
    \item Sein Selbst. \item Seine Herrschaft.
  \end{itemize}
\item Wenn beide Namen al-Ḥayy (Der ewig Lebende) und al-Qayyūm (Der Beständige) verknüpft
  werden, dann beweist das die Vollkommenheit sowohl Seines Selbst als auch Seiner Herrschaft.
  (richtig -- falsch)
\item Die Verknüpfung beider Namen Allahs, al-Ḥayy (Der ewig Lebende) und al-Qayyūm (Der
  Beständige), wiederholt sich im Qur’ān an:
  \begin{itemize}
    \item drei Stellen. \item vier Stellen. \item zwei Stellen.
  \end{itemize}
\item Was die verneinten Eigenschaften angeht, so müssen wir sie für Allah bestreiten, wie Er sie
  für Sich und wie Sein Gesandter \ar{ﷺ} für Ihn bestritten haben. Dabei müssen wir aber das
  Gegenteil bestätigen, denn das absolute Bestreiten gehört nicht zur Vollkommenheit, z.\,B.:
  Wir bestreiten für Allah sowohl den Schlummer als auch den Schlaf, weil Seine Lebendigkeit
  und Seine Beständigkeit vollkommen sind. (richtig -- falsch)
\item Allah erlaubt niemandem, eine Fürsprache vorzulegen, außer wenn sie sich auf etwas
  bezieht, das Ihm wohlgefällig ist. Bekanntlich ist Ihm nur Folgendes wohlgefällig:
  \begin{itemize}
    \item der monotheistische Glaube.
    \item das Folgen Seiner Gesandten.
    \item beide vorigen Aspekte.
  \end{itemize}
\item Was Allah Seinen Geschöpfen an Angelegenheiten der Religion und des Schicksals
  mitgeteilt hat, ist:
  \begin{itemize}
    \item wenig. \item viel.
  \end{itemize}
\item „Er weiß, was vor ihnen“ bezieht sich auf das Gegenwärtige und Künftige, während „was
  hinter ihnen liegt“ das Vergangene betrifft. (richtig -- falsch)
\item Allahs Name al-ʿAliyy (der Erhabene) bedeutet:
  \begin{itemize}
    \item an sich.
    \item aufgrund Seiner Eigenschaften.
    \item Der all Seine Geschöpfe überwältigt hat.
    \item all die vorigen Aspekte.
  \end{itemize}
\item Wer diesen Vers in einer Nacht rezitiert, bei dem bleibt ein Hüter von Allah, so dass kein
  Teufel sich ihm bis zum Morgen nähern kann. Welcher ist dieser Vers?
  \begin{itemize}
    \item Der letzte Vers in Sūrat al-Baqara (2:286, Die Kuh).
    \item Āyat al-Kursiyy (2:255, Der Thronvers).
  \end{itemize}
\item Der Thronvers soll rezitiert werden:
  \begin{itemize}
    \item im Anschluss an die Pflichtgebete.
    \item wenn man sich ins Bett legt.
    \item am Morgen und am Abend.
    \item zu all diesen Zeiten.
  \end{itemize}
\end{enumerate}

  \end{itemize}

  \begin{itemize}

  \end{itemize}

\subsection*{Fragen zur Exegese von Sūrat at-Takāṯur (102, Das Vermehren)}

\begin{enumerate}[leftmargin=1.6em,resume]
\item Diese Sūra ist:
  \begin{itemize}
    \item mekkanisch. \item medinesisch.
  \end{itemize}
\item Diese Sūra impliziert:
  \begin{itemize}
    \item eine Mitteilung über den Zustand der Menschen.
    \item Tadeln der Menschen, weil sie sich mit anderen Sachen beschäftigen, als das Ziel,
      wofür sie erschaffen wurden.
  \end{itemize}
\item In dieser Sūra wird das Vermehren verboten, auch wenn es um Allahs Willen wäre.
  (richtig -- falsch)
\item Der Ausdruck „Jemand ist zu seiner letzten (Ruhe-)Stätte verlegt worden“ (als
  euphemistische Umschreibung des Todes):
  \begin{itemize}
    \item impliziert eine Verleugnung der Auferstehung. \item ist legitim.
  \end{itemize}
\item In dem Ausdruck „bis ihr die Friedhöfe besucht“ (102:2) bezeichnet Allah die Toten als
  Besucher und nicht als Bewohner der Friedhöfe. Warum?
  \begin{itemize}
    \item Weil die Zwischenwelt (al-Barzaḫ) eine Übergangswelt zum Jenseits ist.
    \item Weil sie ihre diesseitigen Häuser zu den Friedhöfen verließen, wobei diese nicht ihre
      eigenen Häuser sind.
  \end{itemize}
\item Das Wissen wird in drei Arten eingeteilt: Kenntnis der Gewissheit, Auge der Gewissheit,
  Recht der Gewissheit. (richtig -- falsch)
\end{enumerate}

  \end{itemize}

  \begin{itemize}

    \item Eine Warnung sowohl vor jenem Höllental als auch vor Anderem.
  \end{itemize}

  \end{itemize}

  \end{itemize}

    \item die Ungläubigen in Mekka.
  \end{itemize}
\item Diese Sūra impliziert die tatsächliche Distanzierung von der Beigesellung und deren
  Täter, und zwar mit dem Herzen (d.\,h. mit dem Glauben), der Zunge (d.\,h. mit den Worten)

  \begin{itemize}
    \item die Bestätigung.
    \item die erste Wiederholung weist darauf hin, dass die Tätigkeit des Anbetens hier
      bestritten wird, während die zweite Wiederholung diese Einschätzung bestätigt.
  \end{itemize}
\end{enumerate}
