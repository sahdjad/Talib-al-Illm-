  \AddToShipoutPictureBG*{%
    \AtPageLowerLeft{\color{Lesson#2}\rule{\paperwidth}{\paperheight}}%
  }
  \null\vspace*{0.30\textheight}
  \begin{center}
    {\headingfont\color{OffWhite!85!Lesson#2}\fontsize{22}{26}\selectfont LEKTION}\\[2mm]
    {\headingfont\bfseries\color{OffWhite}\fontsize{72}{80}\selectfont #1}\\[10mm]
    {\headingfont\bfseries\color{OffWhite}\fontsize{20}{24}\selectfont #3}
  \end{center}
  \clearpage
}
